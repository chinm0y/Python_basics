

# 1. BASIC ARITHMETIC OPERATORS
print("="*60)
print("1. BASIC ARITHMETIC OPERATORS")
print("="*60)

a = 20
b = 3

# Addition (+)
print(f"\nAddition: {a} + {b} = {a + b}")

# Subtraction (-)
print(f"Subtraction: {a} - {b} = {a - b}")

# Multiplication (*)
print(f"Multiplication: {a} * {b} = {a * b}")

# Division (/) - Returns float
print(f"Division: {a} / {b} = {a / b}")
print(f"Division: {a} / {b} = {a / b:.2f}")

# Floor Division (//) - Returns integer
print(f"Floor Division: {a} // {b} = {a // b}")

# Modulus (%) - Returns remainder
print(f"Modulus: {a} % {b} = {a % b}")

# Exponentiation (**)
print(f"Exponentiation: {a} ** {b} = {a ** b}")


# 2. COMPLEX CALCULATIONS
print("\n" + "="*60)
print("2. COMPLEX CALCULATIONS")
print("="*60)

x = 15
y = 4
z = 2

# Order of operations (PEMDAS)
result1 = x + y * z
result2 = (x + y) * z
result3 = x ** y + z

print(f"\n15 + 4 * 2 = {result1} (Multiplication first)")
print(f"(15 + 4) * 2 = {result2} (Parentheses first)")
print(f"15 ** 4 + 2 = {result3}")

# 4. PRACTICAL EXAMPLES

print("4. PRACTICAL EXAMPLES")
print("="*60)

# Example 1: Calculate average
marks = [85, 90, 78, 92, 88]
average = sum(marks) / len(marks)
print(f"\nMarks: {marks}")
print(f"Average: {average:.2f}")

# Example 2: Calculate area and perimeter
radius = 5
pi = 3.14159
area = pi * radius ** 2
circumference = 2 * pi * radius
print(f"\nCircle with radius {radius}:")
print(f"Area: {area:.2f}")
print(f"Circumference: {circumference:.2f}")

# Example 3: Simple calculator
print("\nSimple Calculator:")
num1 = 100
num2 = 8
print(f"Number 1: {num1}, Number 2: {num2}")
print(f"Sum: {num1 + num2}")
print(f"Difference: {num1 - num2}")
print(f"Product: {num1 * num2}")
print(f"Quotient: {num1 / num2:.2f}")
print(f"Remainder: {num1 % num2}")
print(f"Power: {num1} to the power of 2 = {num1 ** 2}")

# Example 4: Compound calculations
print("\nCompound Calculation Example:")
principal = 1000
rate = 5
time = 3
simple_interest = (principal * rate * time) / 100
total_amount = principal + simple_interest
print(f"Principal: ${principal}")
print(f"Simple Interest (5% for 3 years): ${simple_interest}")
print(f"Total Amount: ${total_amount}")

# Example 5: String and list operations with multiplication
print("\nRepeating operations:")
text = "Python "
print(f"'{text}' * 3 = '{text * 3}'")

list_items = [1, 2, 3]
print(f"{list_items} * 2 = {list_items * 2}")

# Division by operations
print(f"\n10 / 3 = {10 / 3} (Float result)")
print(f"10 // 3 = {10 // 3} (Integer result)")
print(f"10 % 3 = {10 % 3} (Remainder)")

# Negative numbers
print(f"\n-20 / 3 = {-20 / 3}")
print(f"-20 // 3 = {-20 // 3}")
print(f"-20 % 3 = {-20 % 3}")

# Large numbers
print(f"\n10 ** 100 = {10 ** 100} (Very large number!)")

# Zero cases
print(f"\n0 * 1000 = {0 * 1000}")
print(f"0 / 5 = {0 / 5}")
try:
    result = 10 / 0
except ZeroDivisionError:
    print(f"10 / 0 = Error! Cannot divide by zero")
